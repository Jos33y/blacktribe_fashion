# Search engines and AI assistants are welcome on the shop, collections,
# lookbook and magazine. Private and transactional pages stay out.
#
# Note: a crawler that has its own group below ignores the "*" group,
# so every group repeats the same rules. The more specific line wins,
# so "Allow: /api/products" beats "Disallow: /api/".

# ═══ Everyone ═══

Allow: /
Disallow: /admin
Disallow: /api/
# ...except the read-only data the pages are built from. Google renders the
# shop with these; blocking them made /shop look empty (a "soft 404").
Allow: /api/products
Allow: /api/collections
Allow: /api/categories
Allow: /api/homepage
Allow: /api/editorial
Disallow: /auth
Disallow: /account
Disallow: /checkout
Disallow: /pay/
# Print versions of magazine issues duplicate the web pages
Disallow: /editorial/*/print
